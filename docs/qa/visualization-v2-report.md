.venv/bin/python scripts/visualization_v2_gate.py --write \
  --browser-results /tmp/muta-v2-browser-results.json \
  --revision edb6c93ea0901146911a8c0762d8fc247f633158 --pre-holdout-candidate-sha ae3cc2e1572b042a0d53ab5af4fe1143e9cb71cd --pre-holdout-frozen-at 2026-08-27T12:07:54+01:00
